import java.io.*;
import java.util.*;

public class Main {
    static final long MOD = 1_000_000_007L;

    static int n, k;
    static int[] a;
    static Set<Integer> good;
    static List<Integer>[] tree;

    static Map<Integer, Long> dfs(int v, int parent) {
        Map<Integer, Long> dp = new HashMap<>();
        dp.put(a[v], 1L);

        for (int u : tree[v]) {
            if (u == parent) continue;

            Map<Integer, Long> child = dfs(u, v);
            Map<Integer, Long> next = new HashMap<>();

            for (Map.Entry<Integer, Long> e1 : dp.entrySet()) {
                int xor1 = e1.getKey();
                long ways1 = e1.getValue();

                for (Map.Entry<Integer, Long> e2 : child.entrySet()) {
                    int xor2 = e2.getKey();
                    long ways2 = e2.getValue();
                    long ways = (ways1 * ways2) % MOD;

                    // Option 1: keep the edge
                    int merged = xor1 ^ xor2;
                    next.put(merged,
                            (next.getOrDefault(merged, 0L) + ways) % MOD);

                    // Option 2: cut the edge (child must be valid alone)
                    if (good.contains(xor2)) {
                        next.put(xor1,
                                (next.getOrDefault(xor1, 0L) + ways) % MOD);
                    }
                }
            }
            dp = next;
        }
        return dp;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);
        StringBuilder out = new StringBuilder();

        int t = fs.nextInt();
        while (t-- > 0) {
            n = fs.nextInt();
            k = fs.nextInt();

            tree = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) tree[i] = new ArrayList<>();

            for (int i = 0; i < n - 1; i++) {
                int u = fs.nextInt();
                int v = fs.nextInt();
                tree[u].add(v);
                tree[v].add(u);
            }

            a = new int[n + 1];
            for (int i = 1; i <= n; i++) a[i] = fs.nextInt();

            good = new HashSet<>();
            for (int i = 0; i < k; i++) good.add(fs.nextInt());

            Map<Integer, Long> res = dfs(1, 0);
            long ans = 0;
            for (int b : good) {
                ans = (ans + res.getOrDefault(b, 0L)) % MOD;
            }

            out.append(ans).append('\n');
        }

        System.out.print(out);
    }

    // Fast input
    static class FastScanner {
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;
        private final InputStream in;

        FastScanner(InputStream in) {
            this.in = in;
        }

        int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }

        int nextInt() throws IOException {
            int c, sgn = 1, res = 0;
            do c = read(); while (c <= ' ');
            if (c == '-') {
                sgn = -1;
                c = read();
            }
            while (c > ' ') {
                res = res * 10 + (c - '0');
                c = read();
            }
            return res * sgn;
        }
    }
}
