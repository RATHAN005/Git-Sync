    public List<Long> mergeAdjacent(int[] nums) {
        List<Long> stack = new ArrayList<>();

        for(int num : nums){
            long value = num;

            while(!stack.isEmpty() && stack.get(stack.size()-1)== value){
                value *= 2;
                stack.remove(stack.size() -1);
            }
            if(value != 0){
                stack.add(value);
            }
        }
        return stack;
    }
}
